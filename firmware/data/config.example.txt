# Magnetometer hard-iron offsets, in raw LSB at the 8G field range. Optional:
# omit BOTH keys to use the values compiled into main.cpp. Setting only one is
# ignored - half a hard-iron correction is worse than none.
#
# A calibration describes one rigid assembly, so it has to be redone after any
# physical change - a different breadboard or base, a moved wire, the final
# enclosure. That is why these live here: recalibrating then costs a
# `pio run -t uploadfs` rather than a rebuild and reflash.
#
# Measure them, don't guess. From firmware/:
#     pio run -e mag_diag -t upload
# then from scripts/:
#     python3 -m gustik_scripts.mag_calibrate --tumble
# and paste the two lines it prints. offsetY comes out already negated into
# the firmware's mount frame (up=-z, forward=+x), so paste it verbatim.
# Reflash the station firmware afterwards: pio run -e esp32dev -t upload
#
# mag.offsetX=1713.5
# mag.offsetY=1984.0

# Status LED panel (nine LEDs + one button on the breadboard). Both keys
# optional - omit them for an enabled panel that sleeps to a heartbeat after
# five minutes.
#
# leds.enabled=false leaves all nine pins low from boot and ignores the
# button. It is the durable way to silence the panel; a long press on the
# button also blanks it, but only until the next power cycle.
#
# leds.timeoutSeconds is how long with no button press before the panel
# sleeps to a blue heartbeat (plus the red fault code, if any). 0 = never
# sleep. Sleeping costs nothing visually - "the whole row dark" still means
# the station is not running.
#
# leds.enabled=true
# leds.timeoutSeconds=300
